    import csv
from dataclasses import dataclass


@dataclass
class Instance:
    name: str
    tags: list
    n: int
    edges: list  # (u, v, wu, wv): wu = weight if oriented toward u, wv = toward v


def _num(s):
    x = float(s)
    return int(x) if x.is_integer() and "." not in s and "e" not in s.lower() else x


def load_text(path):
    """Read the .txt export: comment lines, then "n m", then m lines "u v wu wv"."""
    with open(path) as f:
        rows = f.read().splitlines()
    out, name, tags, i = [], "", [], 0
    while i < len(rows):
        s = rows[i].strip()
        i += 1
        if not s:
            continue
        if s.startswith("#"):
            body = s[1:].strip()
            if body.startswith("instance:"):
                name = body[len("instance:"):].strip()
            elif body.startswith("tags:"):
                tags = [t.strip() for t in body[len("tags:"):].split(",") if t.strip()]
            continue
        n, m = map(int, s.split())
        edges = []
        while len(edges) < m:
            line = rows[i].strip()
            i += 1
            if not line or line.startswith("#"):
                continue
            u, v, wu, wv = line.split()
            edges.append((int(u), int(v), _num(wu), _num(wv)))
        out.append(Instance(name, tags, n, edges))
        name, tags = "", []
    return out


def load_csv(path):
    """Read the .csv export: one row per edge (an instance with no edges has one row with empty u,v,wu,wv)."""
    groups = {}
    with open(path, newline="") as f:
        for r in csv.DictReader(f):
            g = groups.setdefault(
                r["instance"],
                Instance(r["instance"], [t for t in r["tags"].split(";") if t], int(r["n"]), []),
            )
            if r["u"] != "":
                g.edges.append((int(r["u"]), int(r["v"]), _num(r["wu"]), _num(r["wv"])))
    return list(groups.values())
